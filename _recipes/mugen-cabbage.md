<div class="lang" data-lang="en">
  <h1 class="title-recipe">🥬 Yamitsuki Cabbage</h1>
  <div class="container">
    <p><strong>Yield:</strong> Serves 2</p>
  </div>

  <div class="container">
    <h2 class="title-ingredient">Ingredients</h2>

    <div class="ingredient-quantity">
      <div class="column-left-ingredient">Cabbage</div>
      <div class="column-right-amount">1/4 head</div>
    </div>
    <div class="ingredient-quantity">
      <div class="column-left-ingredient">Canned tuna</div>
      <div class="column-right-amount">1 can</div>
    </div>
    <div class="ingredient-quantity">
      <div class="column-left-ingredient">Mayonnaise</div>
      <div class="column-right-amount">1 tsp</div>
    </div>
    <div class="ingredient-quantity">
      <div class="column-left-ingredient">Sesame oil</div>
      <div class="column-right-amount">1 tsp</div>
    </div>
    <div class="ingredient-quantity">
      <div class="column-left-ingredient">Chicken bouillon powder</div>
      <div class="column-right-amount">1 tsp</div>
    </div>
    <div class="ingredient-quantity">
      <div class="column-left-ingredient">Pepper</div>
      <div class="column-right-amount">A little</div>
    </div>
  </div>

  <div class="container">
    <h2 class="title-instruction">Instructions</h2>
    <ol>
      <li>Finely shred the cabbage.</li>
      <li>Put all ingredients except the pepper in a microwave-safe bowl and mix.</li>
      <li>Cover loosely with plastic wrap and microwave at 600W for 2 minutes.</li>
      <li>Stir, sprinkle with pepper, and serve.</li>
    </ol>
  </div>
</div>
